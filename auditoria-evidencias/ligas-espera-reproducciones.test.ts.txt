import { beforeEach, expect, it, vi } from "vitest"
const mocks = vi.hoisted(() => ({ gate: vi.fn(), client: vi.fn() }))
vi.mock("@/lib/serverLeagueAccess", () => ({ getServerLeagueActor: mocks.gate }))
vi.mock("@/lib/supabaseServer", () => ({ createSupabaseServiceClient: mocks.client }))
const leagueId = "11111111-1111-4111-8111-111111111111"
const seasonId = "22222222-2222-4222-8222-222222222222"
beforeEach(() => vi.clearAllMocks())

it("BUG-006: un jugador tercero en espera recibe position=1", async () => {
  const rows = [{ user_id: "first", status: "waiting", position: 1 }, { user_id: "second", status: "waiting", position: 2 }, { user_id: "qa-user", status: "waiting", position: 3 }]
  const supabase = { from: (table: string) => {
    let userFilter: string | null = null
    const query: any = { select: () => query, eq: (key: string, value: string) => { if(key === "user_id") userFilter = value; return query }, in: () => query, order: () => query,
      then: (resolve: any) => Promise.resolve({ data: table === "season_waitlist" ? rows.filter(row => !userFilter || row.user_id === userFilter) : [{ id: "qa-user", display_name: "QA" }], error: null }).then(resolve) }
    return query
  } }
  mocks.gate.mockResolvedValue({ ok: true, actor: { supabase, membership: { role: "player" }, user: { id: "qa-user", isSuperuser: false } } })
  const { GET } = await import("@/app/api/leagues/[id]/seasons/[seasonId]/waitlist/route")
  const response = await GET(new Request("http://audit.invalid"), { params: Promise.resolve({ id: leagueId, seasonId }) })
  const payload = await response.json()
  expect(response.status).toBe(200)
  expect(payload.items[0].position).toBe(3)
  expect(payload.position).toBe(1)
})

it("BUG-007: volver a unirse conserva fila cancelled por ignoreDuplicates", async () => {
  const existing = { league_id: leagueId, season_id: seasonId, user_id: "qa-user", status: "cancelled", position: 2 }
  let upsertOptions: any
  const query: any = { select: () => query, eq: () => query, order: () => query, limit: () => query,
    maybeSingle: async () => ({ data: null, error: null }),
    upsert: (value: any, options: any) => { upsertOptions = options; if(!options.ignoreDuplicates) Object.assign(existing,value); return query } }
  const { joinSeasonWaitlist } = await import("@/lib/serverSeasonWaitlist")
  const result = await joinSeasonWaitlist({ supabase: { from: () => query }, leagueId, seasonId, userId: "qa-user" })
  expect(upsertOptions.ignoreDuplicates).toBe(true)
  expect(existing.status).toBe("cancelled")
  expect(result).toBeNull()
})

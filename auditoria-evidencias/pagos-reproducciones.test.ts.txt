 const matchId="11111111-1111-4111-8111-111111111111"
 const booking=buildCourtBooking({participantIds:["a","b","c","d"],reservations:[{playerId:"a",amount:40}],ballPurchases:[]})

 const {PUT}=await import("@/app/api/matches/[matchId]/court-booking/transfers/[transferId]/route")

 expect(responses.map(r=>r.status)).toEqual([200,200])

it("BUG-029: reparto de centimos puede dejar diferencia de dos centimos entre jugadores",()=>{
 const ids=["a","b","c","d"]
 const booking=buildCourtBooking({participantIds:ids,reservations:[{playerId:"a",amount:22.06}],ballPurchases:[]})
 const costs=new Map(ids.map(id=>[id,id==="a"?2206:0]))
 for(const t of booking.transfers){costs.set(t.fromPlayerId,costs.get(t.fromPlayerId)!+Math.round(t.amount*100));costs.set(t.toPlayerId,costs.get(t.toPlayerId)!-Math.round(t.amount*100))}
 const values=[...costs.values()];expect(values.reduce((a,b)=>a+b,0)).toBe(2206)
 expect(Math.max(...values)-Math.min(...values)).toBe(2)
})
